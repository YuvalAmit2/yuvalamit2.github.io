% ---------------------------------------------------------------
% \cvline{left}{right}
%   Left text, a dot leader, and right-aligned text (usually a date).
%   If the left text is too long it wraps onto further lines
%   (ragged right), and the dots + date sit on the last line.
%   There are always at least 2em of dots, so nothing can overflow.
% ---------------------------------------------------------------
\newcommand{\cvline}[2]{%
  {\rightskip=0pt plus 4em\relax\parshape=0 \hangindent=1em\hangafter=1\relax
   #1\unskip\nobreak\hspace{0.5em}\nobreak
   \leaders\hbox to 0.45em{\hss.\hss}\hskip 2em plus 1fill\relax
   \nobreak\hspace{0.5em}\mbox{#2}\par}%
}

% Indented description paragraph below an entry (skipped if empty)
\newcommand{\cvdesc}[1]{%
  \if\relax\detokenize{#1}\relax\else
    {\vspace{1pt}\leftskip=1em\noindent #1\par}%
  \fi
}

% \boldentry{Title}{Date}{Description}
\newcommand{\boldentry}[3]{%
  \item \cvline{{#1}}{#2}\cvdesc{#3}%
}

% \talk{Talk title}{Venue}{Date}
%   Title and date on the first line, venue indented below.
\newcommand{\talk}[3]{%
  \item \cvline{#2}{#3}\cvdesc{\textit{#1}}%
  \vspace{2pt}
}

% \twopart{Role or award}{Organization}{Date}
\newcommand{\twopart}[3]{%
  \item \cvline{#1, #2}{#3}%
}

% ---------------------------------------------------------------
% Papers, read from a .bib file (\addbibresource{...} in cv.tex)
%
%   Preprint:  Title [arXiv, year]
%   Published: Title [arXiv] [Journal, year]
%              With coauthors (smaller, gray)
%
%   arXiv link:   from the eprint field (arxiv.org/abs/<eprint>)
%   Journal link: from the doi field (doi.org/<doi>), if there is one
%   A paper counts as published when it has a journal field.
%   Coauthors: every author except me (\cvmegiven, \cvmefamily below).
% ---------------------------------------------------------------
\usepackage[backend=biber, sorting=none, maxnames=999, minnames=999]{biblatex}

\newcommand{\cvmegiven}{Yuval}
\newcommand{\cvmefamily}{Amit}

% Each paper is one cvlist item; papers appear in .bib file order
\defbibenvironment{cvpapers}{\begin{cvlist}}{\end{cvlist}}{\item}

% Names: count / print every author except me, comma separated
\newcounter{cvcoauthors}
\newcommand{\cvifme}[2]{%
  \ifboolexpr{test {\ifdefstrequal{\namepartgiven}{\cvmegiven}} and
              test {\ifdefstrequal{\namepartfamily}{\cvmefamily}}}{#1}{#2}}
\DeclareNameFormat{cvcount}{\cvifme{}{\stepcounter{cvcoauthors}}}
\DeclareNameFormat{cvcoauthors}{%
  \cvifme{}{%
    \ifnumgreater{\value{cvcoauthors}}{0}{, }{}%
    \stepcounter{cvcoauthors}%
    \mbox{\namepartgiven\space\namepartfamily}}}

\newbibmacro{cvpaper}{%
  % Line 1: title and links
  {\rightskip=0pt plus 4em\relax\parshape=0 \hangindent=1em\hangafter=1\relax
   \textit{\thefield{title}}%
   \iffieldundef{eprint}{}{%
     \space[\href{https://arxiv.org/abs/\thefield{eprint}}{arXiv}%
     \iffieldundef{journaltitle}{, \thefield{year}}{}]}%
   \iffieldundef{journaltitle}{}{%
     \space[\iffieldundef{doi}
       {\thefield{journaltitle}}
       {\href{https://doi.org/\thefield{doi}}{\thefield{journaltitle}}}%
     , \thefield{year}]}%
   \par}%
  % Line 2: coauthors (skipped for a single-author paper)
  \setcounter{cvcoauthors}{0}%
  \printnames[cvcount]{author}%
  \ifnumgreater{\value{cvcoauthors}}{0}{%
    \setcounter{cvcoauthors}{0}%
    \cvdesc{\small\color{cvgray}\rightskip=0pt plus 1fil\relax
      With \printnames[cvcoauthors]{author}\par}}{}%
}
\DeclareBibliographyDriver{misc}{\usebibmacro{cvpaper}}
\DeclareBibliographyDriver{article}{\usebibmacro{cvpaper}}
